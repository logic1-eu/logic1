>>> from logic1 import *
>>> from logic1.theories.Sets import *

>>> a, b, c, d = VV.get('a', 'b', 'c', 'd')

>>> dnf(T)
T

>>> dnf(F)
F

>>> dnf(And(T, T))
T

>>> dnf(a == d)
a == d

>>> dnf(Equivalent(a == d, b == d))
Or(a == b, And(a != d, b != d))

>>> dnf(((a == d) | (b != d)) & ((a != d) | (b == d)))
Or(a == b, And(a != d, b != d))

>>> dnf(((a != d) | (b == d)) & ((a == d) | (b == d)))
Or(And(a == b, a == d), And(b == d, a != b))

>>> cnf(T)
T

>>> cnf(F)
F

>>> cnf(And(T, T))
T

>>> cnf(a == d)
a == d

>>> cnf(Equivalent(a == d, b == d))
And(Or(a == b, a != d), Or(a == b, b != d))

>>> cnf(((a == d) | (b != d)) & ((a != d) | (b == d)))
And(Or(a == b, a != d), Or(a == b, b != d))

>>> cnf(((a != d) | (b == d)) & ((a == d) | (b == d)))
And(Or(a == b, a != d), Or(a == d, b == d))

The Boolean abstraction is shared by both entry points but reset after each
conversion:

>>> xs = VV.get('x0', 'x1', 'x2', 'x3', 'x4', 'x5', 'x6', 'x7')
>>> for left, right in zip(xs[::2], xs[1::2]):
...     _ = dnf(And(left == d, right == d))
>>> converter = dnf.__self__
>>> converter is cnf.__self__
True
>>> converter._index
0
>>> converter._atoms_to_pyeda
{}
>>> converter._pyeda_to_atoms
{}
